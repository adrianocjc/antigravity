Atividade 1: Card de Perfil Dinâmico

Criação de um componente visual clássico de card de perfil de usuário. Ideal para ensinar manipulação de layouts modernos, tipografia, bordas arredondadas e transições CSS de alta performance.

Objetivos Educacionais
  • Dominar o uso de tags semânticas do HTML5 (`<article>`, `<header>`, `<footer>`).
  • Compreender o alinhamento flexível e responsivo com **CSS Flexbox**.
  • Implementar variáveis CSS para gerenciar cores de maneira escalável.
  • Criar interações dinâmicas e transições fluidas com pseudo-classes (`:hover`, `:focus`).
Instruções Passo a Passo
  1. Crie a estrutura básica HTML contendo a foto do perfil, nome, profissão, links sociais e biografia.
  2. Aplique reset de CSS padrão para garantir consistência entre navegadores.
  3. Centralize o card na tela usando Flexbox no elemento pai.
  4. Estilize o card com fundo translúcido, cantos arredondados, sombras sutis e transição hover que aumenta o card ligeiramente (`transform: translateY(-5px)`).

// Seletor de Elementos
const todoForm = document.getElementById('todo-form');
const todoInput = document.getElementById('todo-input');
const todoList = document.getElementById('todo-list');
const taskCounter = document.getElementById('task-counter');
const clearCompletedBtn = document.getElementById('clear-completed');
const filterBtns = document.querySelectorAll('.filter-btn');

// Estado Global da Aplicação
let tasks = JSON.parse(localStorage.getItem('my_academy_tasks')) || [];
let currentFilter = 'all';

// Função: Atualizar Contador de Pendentes
function updateCounter() {
    const pendings = tasks.filter(t => !t.completed).length;
    taskCounter.textContent = `${pendings} ${pendings === 1 ? 'pendente' : 'pendentes'}`;
}

// Função: Persistir no LocalStorage
function saveToStorage() {
    localStorage.setItem('my_academy_tasks', JSON.stringify(tasks));
}

// Função: Renderizar Lista na Tela
function renderTasks() {
    // Limpar lista existente
    todoList.innerHTML = '';

    // Filtrar tarefas baseado na seleção ativa
    const filteredTasks = tasks.filter(task => {
        if (currentFilter === 'pending') return !task.completed;
        if (currentFilter === 'completed') return task.completed;
        return true;
    });

    if (filteredTasks.length === 0) {
        todoList.innerHTML = `
  • Nenhuma tarefa encontrada.
    • `;
        updateCounter();
        return;
    }

    filteredTasks.forEach(task => {
        const li = document.createElement('li');
        li.className = `todo-item ${task.completed ? 'completed' : ''}`;

        li.innerHTML = `
            
    
                
                ${escapeHTML(task.text)}
            
    
            
        `;
        todoList.appendChild(li);
    });

    updateCounter();
}

// Helper para prevenir injeções de HTML maliciosas (XSS)
function escapeHTML(text) {
    const div = document.createElement('div');
    div.innerText = text;
    return div.innerHTML;
}

// Função: Adicionar Nova Tarefa
todoForm.addEventListener('submit', (e) => {
    e.preventDefault();
    const taskText = todoInput.value.trim();
    if (!taskText) return;

    const newTask = {
        id: Date.now(),
        text: taskText,
        completed: false
    };

    tasks.push(newTask);
    todoInput.value = '';
    
    saveToStorage();
    renderTasks();
});

// Função: Alternar Status Concluído/Pendente
window.toggleTaskStatus = function(id) {
    tasks = tasks.map(task => {
        if (task.id === id) {
            return { ...task, completed: !task.completed };
        }
        return task;
    });
    saveToStorage();
    renderTasks();
};

// Função: Deletar Tarefa
window.deleteTask = function(id) {
    tasks = tasks.filter(task => task.id !== id);
    saveToStorage();
    renderTasks();
};

// Função: Limpar Todas as Concluídas
clearCompletedBtn.addEventListener('click', () => {
    tasks = tasks.filter(task => !task.completed);
    saveToStorage();
    renderTasks();
});

// Manipulador dos Botões de Filtro
filterBtns.forEach(btn => {
    btn.addEventListener('click', (e) => {
        filterBtns.forEach(b => b.classList.remove('active'));
        e.target.classList.add('active');
        currentFilter = e.target.getAttribute('data-filter');
        renderTasks();
    });
});

// Inicialização Inicial
renderTasks();

// Função: Renderizar Produtos no Grid
function renderCatalog(filterText = '') {
    productGrid.innerHTML = '';
    
    // Filtrar produtos baseados no texto de busca (case-insensitive)
    const filtered = products.filter(p => 
        p.name.toLowerCase().includes(filterText.toLowerCase())
    );

    if (filtered.length === 0) {
        productGrid.innerHTML = `
    Nenhum periférico encontrado.
    `;
        return;
    }

    filtered.forEach(p => {
        const card = document.createElement('article');
        card.className = 'product-card';
        card.innerHTML = `
            
            
    ${p.name}
    
            
    R$ ${p.price.toFixed(2).replace('.', ',')}
    
            
        `;
        productGrid.appendChild(card);
    });
}

// Função: Atualizar painel do Carrinho e Cálculos
function updateCartUI() {
    cartItemsContainer.innerHTML = '';

    if (cart.length === 0) {
        cartItemsContainer.innerHTML = `
    Seu carrinho está vazio.
    `;
        cartCounter.textContent = '0';
        summaryQty.textContent = '0';
        summaryTotal.textContent = 'R$ 0,00';
        checkoutBtn.disabled = true;
        return;
    }

    // Gerar elementos para cada item
    cart.forEach(item => {
        const itemRow = document.createElement('div');
        itemRow.className = 'cart-item';
        itemRow.innerHTML = `
            
    
                ${item.name}
                
    Qty: ${item.qty}
    
            
    
            
    
                R$ ${(item.price * item.qty).toFixed(2).replace('.', ',')}
                
            
    
        `;
        cartItemsContainer.appendChild(itemRow);
    });

    // Calcular valores agregados usando métodos funcionais
    const totalQty = cart.reduce((acc, item) => acc + item.qty, 0);
    const totalPrice = cart.reduce((acc, item) => acc + (item.price * item.qty), 0);

    cartCounter.textContent = totalQty;
    summaryQty.textContent = totalQty;
    summaryTotal.textContent = `R$ ${totalPrice.toFixed(2).replace('.', ',')}`;
    checkoutBtn.disabled = false;
}

// 1. Validar se a requisição é do tipo POST
if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    // Redireciona de volta em caso de acessos indevidos
    header('Location: index.php');
    exit;
}

// 2. Importar conexão do BD
require_once 'config.php';

// 3. Capturar e Sanitizar Dados (Combate a dados sujos de entrada)
$nome = filter_input(INPUT_POST, 'nome', FILTER_SANITIZE_SPECIAL_CHARS);
$email = filter_input(INPUT_POST, 'email', FILTER_VALIDATE_EMAIL);
$mensagem = filter_input(INPUT_POST, 'mensagem', FILTER_SANITIZE_SPECIAL_CHARS);

// Tratar falhas de sanitização/validação
if (!$nome || !$email || !$mensagem) {
    die("Falha de Validação: Certifique-se de preencher todos os campos corretamente.");
}

try {
    // 4. Preparação da Query SQL com Placeholders (Impede Injeção SQL)
    $sql = "INSERT INTO feedbacks (nome, email, mensagem) VALUES (:nome, :email, :mensagem)";
    $stmt = $pdo->prepare($sql);

    // 5. Vincular parâmetros e Executar com os dados higienizados
    $stmt->execute([
        ':nome'     => $nome,
        ':email'    => $email,
        ':mensagem' => $mensagem
    ]);

    // 6. Redirecionamento com sinal de Sucesso
    header('Location: index.php?sucesso=1');
    exit;

} catch (PDOException $e) {
    // Em produção, nunca exiba detalhes do banco de dados. Salve em logs.
    die("Erro grave de banco de dados ao salvar feedback: " . $e->getMessage());
}
